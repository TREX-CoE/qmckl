void floatingPointExceptionHandler(int signal) {
    void* backtraceArray[MAX_BACKTRACE_SIZE];
    int backtraceSize = backtrace(backtraceArray, MAX_BACKTRACE_SIZE);
    char** backtraceSymbols = backtrace_symbols(backtraceArray, backtraceSize);

* Introduction

 The API lets the caller state the number of bits of significand (~precision~)
 and of exponent (~range~) it needs, so that an implementation can change its
 arithmetic without changing the API. The default is IEEE 754 double
 precision, 53 bits of significand and 11 bits of exponent.

 In the reference implementation the precision sets the machine epsilon
 $\epsilon = 2^{1-n}$ and the thresholds that follow from it: the exponent at
 which the exponential of a Gaussian primitive is dropped, where the evaluation
 of an orbital stops, and the smallest Jastrow term that a sparse product
 keeps. In an ~--enable-hpc~ build the same value chooses between single and
 double precision: the atomic-orbital kernels evaluate the primitive
 exponentials in single precision at 23 bits or less, and the molecular-orbital
 kernels switch to their single-precision variants at 24 bits or less for the
 values and 26 bits or less for the VGL. The range is stored and returned; no
 kernel reads it.

 ~qmckl_get_numprec_epsilon~ returns the epsilon of the requested precision,
 and ~qmckl_test_precision_64~ compares two doubles and returns how many of
 their 53 significand bits agree. With ~--enable-fpe~ the library enables the
 invalid, division-by-zero and overflow floating-point exceptions when it is
 loaded, and installs a ~SIGFPE~ handler that prints a backtrace and exits.

* Control of the numerical precision

  /Precision/ is the number of significand bits, the implicit leading bit
  included, and /range/ the number of exponent bits, as in IEEE 754. The
  context starts with the values that binary64 uses:

  - 53 bits of precision (52 explicit and 1 implicit)
  - 11 bits of exponent

  The setters ~qmckl_set_numprec_precision~ and ~qmckl_set_numprec_range~
  take a context and write one of the two 32-bit unsigned integers that
  ~qmckl_numprec_struct~ holds, under the context lock. They accept a
  precision from 2 to 53 and a range from 2 to 11, and return
  ~QMCKL_INVALID_ARG_2~ outside those bounds, ~QMCKL_INVALID_CONTEXT~ when
  the context is not valid, and ~QMCKL_SUCCESS~ otherwise. The getters
  ~qmckl_get_numprec_precision~ and ~qmckl_get_numprec_range~ return the
  stored value as a 32-bit integer, or an exit code on an invalid context.

* Precision
  ~qmckl_set_numprec_precision~ sets the number of significand bits in the
  context, and ~qmckl_get_numprec_precision~ returns it.

  #+begin_src c :tangle (eval c) :exports none
qmckl_exit_code qmckl_set_numprec_precision(const qmckl_context context, const int precision) {

  if (qmckl_context_check(context) == QMCKL_NULL_CONTEXT)
    return QMCKL_INVALID_CONTEXT;

  if (precision <  2) {
    return qmckl_failwith(context,
                          QMCKL_INVALID_ARG_2,
                          "qmckl_update_numprec_precision",
                          "precision < 2");
  }

  if (precision > 53) {
    return qmckl_failwith(context,
                          QMCKL_INVALID_ARG_2,
                          "qmckl_update_numprec_precision",
                          "precision > 53");
  }

  qmckl_context_struct* const ctx = (qmckl_context_struct*) context;

  /* This should be always true because the context is valid */
  assert (ctx != NULL);

  qmckl_lock(context);
  {
    ctx->numprec.precision = (uint32_t) precision;
  }
  qmckl_unlock(context);

  return QMCKL_SUCCESS;
}

* Helper functions

** Epsilon

   ~qmckl_get_numprec_epsilon~ returns $\epsilon = 2^{1-n}$ where ~n~ is
   the precision. For an invalid context it returns ~NAN~, so that a caller
   cannot mistake an exit code for an epsilon.

   #+begin_src c :comments org :tangle (eval h_func) 
double qmckl_get_numprec_epsilon(const qmckl_context context);
   #+end_src

   # Source
   #+begin_src c :tangle (eval c) :exports none
double qmckl_get_numprec_epsilon(const qmckl_context context) {
  if (qmckl_context_check(context) == QMCKL_NULL_CONTEXT)
    return NAN;
  const qmckl_context_struct* const ctx = (qmckl_context_struct*) context;
  const int precision = (int) ctx->numprec.precision;
  return ldexp(1.0, 1 - precision);
}

** Testing the number of unchanged bits

   To check that an approximation keeps a given number of bits, the library
   counts how many leading significand bits two numbers share. It first
   counts by how many units in the last place (ulps) the two differ:
   ~countUlpDifference_64~ reads the doubles as 64-bit integers, sums the
   two distances to zero when the signs differ, and saturates at ~INT64_MAX~
   instead of overflowing.

   #+begin_src c :tangle (eval c) :exports none
int64_t countUlpDifference_64(double a, double b) {

  int64_t xi, yi;
  memcpy(&xi, &a, sizeof xi);
  memcpy(&yi, &b, sizeof yi);

  // Handle sign bit discontinuity: if the signs are different and either value is not zero
  if ((xi < 0) != (yi < 0) && (a != 0.0) && (b != 0.0)) {
    // Use the absolute values and add the distance to zero for both numbers
    const uint64_t distanceToZeroForX = (xi < 0) ? (uint64_t) (INT64_MAX + xi) : (uint64_t) (INT64_MAX - xi);
    const uint64_t distanceToZeroForY = (yi < 0) ? (uint64_t) (INT64_MAX + yi) : (uint64_t) (INT64_MAX - yi);
    const uint64_t result = distanceToZeroForX + distanceToZeroForY;
    return (result > (uint64_t) INT64_MAX) ? INT64_MAX : (int64_t) result;
  }

  // Calculate the difference in their binary representations
  const uint64_t ux = (uint64_t) xi;
  const uint64_t uy = (uint64_t) yi;
  const uint64_t result = (ux > uy) ? ux - uy : uy - ux;
  return (result > (uint64_t) INT64_MAX) ? INT64_MAX : (int64_t) result;
}
   #+end_src

   ~qmckl_test_precision_64~ turns that distance into bits: 53 for two equal
   values, and otherwise 53 minus the number of halvings the distance needs to
   reach zero, with a floor at zero. ~qmckl_test_precision_32~ promotes its two
   floats to double and calls the 64-bit version.

   #+begin_src c :comments org :tangle (eval h_func) 
int32_t qmckl_test_precision_64(double a, double b);
int32_t qmckl_test_precision_32(float a, float b);
   #+end_src

   #+begin_src c :tangle (eval c) :exports none
int32_t qmckl_test_precision_64(double a, double b) {

  int64_t diff = countUlpDifference_64(a,b);

  if (diff == 0) return 53;

  int32_t result = 53;

  for (int i=0 ; i<53 && diff != 0 ; ++i) {
    diff >>= 1;
    result--;
  }

  return result;
}

int32_t qmckl_test_precision_32(float a, float b) {
  return qmckl_test_precision_64( (double) a, (double) b );
}

* Approximate functions

** Exponential

   Two fast exponential functions, adapted from Johan Rade's implementation
   (https://gist.github.com/jrade/293a73f89dfef51da6522428c857802d), based
   on Schraudolph's paper. No kernel calls them at present. Both build a
   floating-point value by biasing its exponent field and clamp the scaled
   input to the range where that works.

        N. Schraudolph, "A Fast, Compact Approximation of the Exponential Function",
        /Neural Computation/ *11*, 853–862 (1999).
        (available at https://nic.schraudolph.org/pubs/Schraudolph99.pdf)

   #+begin_src c :tangle (eval c) :exports none
float fastExpf(float x)
{
  const float a = 12102203.0;
  const float b = 1064986816.0;
  x = a * x + b;

  const float c = 8388608.0;
  const float d = 2139095040.0;
  if (x < c || x > d)
    x = (x < c) ? 0.0f : d;

  uint32_t n = (uint32_t) x;
  memcpy(&x, &n, 4);
  return x;
}


double fastExp(double x)
{
  const double a = 6497320848556798.0;
  const double b = 4606985713057410560.0;
  x = a * x + b;

  const double c = 4503599627370496.0;
  const double d = 9218868437227405312.0;
  if (x < c || x > d)
    x = (x < c) ? 0.0 : d;

  uint64_t n = (uint64_t) x;
  memcpy(&x, &n, 8);
  return x;
}

     #+begin_src c :comments link :tangle (eval c_test) :exports none
/* An exit code is not an epsilon: an invalid context yields NaN. */
{
  const double eps = qmckl_get_numprec_epsilon(QMCKL_NULL_CONTEXT);
  assert(eps != eps);
}

/* Opposite signs: the distance to zero of both operands is summed. The
   sum saturates instead of overflowing a signed 64-bit integer. */
assert(qmckl_test_precision_64(1.0, -1.0) == 0);
assert(qmckl_test_precision_64(0.5, -0.25) == 0);

/* Exponent fields far apart make the sum exceed INT64_MAX. The result
   must saturate there, not wrap to a negative value. */
{
  extern int64_t countUlpDifference_64(double a, double b);
  assert(countUlpDifference_64(4.9406564584124654e-324, -1.7976931348623157e308)
         == INT64_MAX);
}

/* Identical and neighbouring values keep their usual precision. */
assert(qmckl_test_precision_64(1.0, 1.0) == 53);
assert(qmckl_test_precision_64(1.0, 0.0) < 53);

return 0;
}
