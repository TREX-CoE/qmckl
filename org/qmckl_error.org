* Introduction

   Almost every function of the library returns a ~qmckl_exit_code~:
   ~QMCKL_SUCCESS~, which is zero, when the call succeeded, and a non-zero
   code describing the error otherwise. The exceptions are the functions that
   return a value or nothing: the context functions, the lock, the provided
   predicates, the numerical-precision getters, the two
   ~qmckl_test_precision~ helpers and ~qmckl_string_of_error~. An error is
   reported through the return value alone, and the calling code decides what
   it means for it.

   A return code alone does not say which call failed on which argument. For
   that the library records, inside the context, the code, the name of the
   function and a message. This module defines the exit codes and the
   functions that write, read and decode that record.

   The library does not print errors. Assertions catch a misuse in a build
   that keeps them enabled, and ~qmckl_check~, at the end of this page, is the
   only helper that writes to ~stderr~.

* Exit codes

   An exit code is a 32-bit integer:
   #+NAME: type-exit-code
   #+begin_src c :comments org :tangle (eval h_type)
typedef int32_t qmckl_exit_code;
   #+end_src


  #+begin_src f90 :comments org :tangle (eval fh_type) :exports none
      integer  , parameter :: qmckl_exit_code = c_int32_t
  #+end_src

   The codes fall in two blocks. Codes 1 to 20 are ~QMCKL_INVALID_ARG_1~ to
   ~QMCKL_INVALID_ARG_20~ and name the offending argument by its position in
   the signature, the context being the first, so a function that rejects its
   third argument returns ~QMCKL_INVALID_ARG_3~. Codes 101 to 109 describe
   the conditions that are not an argument: a failed allocation, a value that
   was never provided, a set attempted twice. Four of them are defined and
   decoded but never returned by the library at present, ~QMCKL_ERRNO~,
   ~QMCKL_DEALLOCATION_FAILED~, ~QMCKL_OUT_OF_BOUNDS~ and
   ~QMCKL_INVALID_EXIT_CODE~.

* Decoding errors

   ~qmckl_string_of_error~ turns an exit code into a string that an
   application can print or log. The switch covers every code of the table
   and returns ~"Unknown error"~ for anything else, so the result is never
   ~NULL~. Only ~QMCKL_ERRNO~ gives something else: ~strerror(errno)~, the
   description of the last failing system call, in a buffer that the next
   ~strerror~ call may overwrite. The Fortran interface binds to a private
   wrapper, ~qmckl_string_of_error_f~, which copies at most 127 characters
   into the 128-byte buffer it is given and terminates the string.

   #+begin_src c :comments org :tangle (eval h_func) :noweb yes
const char*
qmckl_string_of_error (const qmckl_exit_code error);
   #+end_src

   #+begin_src c :comments org :tangle (eval h_private_func) :exports none :noweb yes
void
qmckl_string_of_error_f(const qmckl_exit_code error,
                        char result[<<MAX_STRING_LENGTH()>>]);
   #+end_src

* Data structure in context

  The context holds one ~qmckl_error_struct~, with the code, a
  ~QMCKL_MAX_FUN_LEN~-byte function name and a ~QMCKL_MAX_MSG_LEN~-byte
  message. The two strings have a fixed size, so recording an error allocates
  nothing and cannot fail for lack of memory.

  A new context is zeroed by ~qmckl_context_create~: it starts with
  ~QMCKL_SUCCESS~ and two empty strings. ~qmckl_context_copy~ copies the
  structure with the rest of the context, and nothing clears it afterwards,
  so a successful call leaves the last error in place.

   #+begin_src c :comments org :tangle (eval h_private_type) :exports none
#define  QMCKL_MAX_FUN_LEN   256
#define  QMCKL_MAX_MSG_LEN  1024

typedef struct qmckl_error_struct {

  qmckl_exit_code exit_code;
  char function[QMCKL_MAX_FUN_LEN];
  char message [QMCKL_MAX_MSG_LEN];

} qmckl_error_struct;
   #+end_src

* Updating errors in the context

   ~qmckl_set_error~ overwrites the record with a code, a function name and a
   message, under the context lock. All three are mandatory and the code
   cannot be ~QMCKL_SUCCESS~. The check that the context is valid is an
   assertion, not a return value, because a record that has nowhere to go
   cannot report anything. The function returns ~QMCKL_SUCCESS~, which says
   that the recording succeeded, not the code it stored. A name longer than
   255 characters and a message longer than 1023 are truncated.

   # Header
   #+begin_src c :comments org :tangle (eval h_func)
qmckl_exit_code
qmckl_set_error(qmckl_context context,
                const qmckl_exit_code exit_code,
                const char* function_name,
                const char* message);
   #+end_src

  # Source
  #+begin_src c :tangle (eval c) :exports none
qmckl_exit_code
qmckl_set_error(qmckl_context context,
                const qmckl_exit_code exit_code,
                const char* function_name,
                const char* message)
{
  /* Passing a function name and a message is mandatory. */
  assert (function_name != NULL);
  assert (message != NULL);

  /* Exit codes are assumed valid. */
  assert (exit_code >= 0);
  assert (exit_code != QMCKL_SUCCESS);
  assert (exit_code < QMCKL_INVALID_EXIT_CODE);

  /* The context is assumed to exist. */
  assert (qmckl_context_check(context) != QMCKL_NULL_CONTEXT);

  qmckl_lock(context);
  {
    qmckl_context_struct* const ctx = (qmckl_context_struct*) context;
    assert (ctx != NULL); /* Impossible because the context is valid. */

    ctx->error.exit_code = exit_code;
    strncpy(ctx->error.function, function_name, QMCKL_MAX_FUN_LEN-1);
    ctx->error.function[QMCKL_MAX_FUN_LEN-1] = '\0';
    strncpy(ctx->error.message, message, QMCKL_MAX_MSG_LEN-1);
    ctx->error.message[QMCKL_MAX_MSG_LEN-1] = '\0';
  }
  qmckl_unlock(context);

  return QMCKL_SUCCESS;
}
    #+end_src

* Get the error

  ~qmckl_get_error~ reads the record back into three caller-provided
  variables. It has no size arguments: the two buffers must hold
  ~QMCKL_MAX_FUN_LEN~ and ~QMCKL_MAX_MSG_LEN~ bytes, and neither may be
  ~NULL~. The code is written to ~exit_code~, which is ~QMCKL_SUCCESS~ when
  nothing has failed since the context was created. Like the setter, it
  asserts that the context is valid and returns ~QMCKL_SUCCESS~.

  # Header
   #+begin_src c :comments org :tangle (eval h_func)
qmckl_exit_code
qmckl_get_error(qmckl_context context,
                qmckl_exit_code *exit_code,
                char* function_name,
                char* message);
   #+end_src

  # Source
  #+begin_src c :tangle (eval c) :exports none
qmckl_exit_code
qmckl_get_error(qmckl_context context,
                qmckl_exit_code *exit_code,
                char* function_name,
                char* message)
{
  /* Passing a function name and a message is mandatory. */
  assert (function_name != NULL);
  assert (message != NULL);

  /* The context is assumed to exist. */
  assert (qmckl_context_check(context) != QMCKL_NULL_CONTEXT);

  qmckl_lock(context);
  {
    qmckl_context_struct* const ctx = (qmckl_context_struct*) context;
    assert (ctx != NULL); /* Impossible because the context is valid. */

    size_t sizeCp;

    sizeCp = strlen(ctx->error.function);
    if (sizeCp >= QMCKL_MAX_FUN_LEN) sizeCp = QMCKL_MAX_FUN_LEN - 1;
    memcpy(function_name, ctx->error.function, sizeCp);
    function_name[sizeCp] = '\0';

    sizeCp = strlen(ctx->error.message);
    if (sizeCp >= QMCKL_MAX_MSG_LEN) sizeCp = QMCKL_MAX_MSG_LEN - 1;
    memcpy(message, ctx->error.message, sizeCp);
    message[sizeCp] = '\0';

    (*exit_code) = ctx->error.exit_code;
  }
  qmckl_unlock(context);

  return QMCKL_SUCCESS;
}

* Failing

   ~qmckl_failwith~ records the failure and returns the code it was given, so
   that a kernel can write ~return qmckl_failwith(...)~. The code, the
   function name and a message are the arguments; a ~NULL~ message is
   replaced by the text ~qmckl_string_of_error~ gives for that code. It
   asserts that the name and the message fit in 255 and 1023 characters. When
   the context is not valid, nothing can be recorded and the function returns
   ~QMCKL_INVALID_CONTEXT~.

   #+begin_src c :comments org :tangle (eval h_func)
qmckl_exit_code
qmckl_failwith(qmckl_context context,
               const qmckl_exit_code exit_code,
               const char* function,
               const char* message) ;
   #+end_src

   #+begin_src c :comments org :tangle (eval c) :exports none
qmckl_exit_code
qmckl_failwith(qmckl_context context,
               const qmckl_exit_code exit_code,
               const char* function,
               const char* message)
{
  assert (exit_code > 0);
  assert (exit_code < QMCKL_INVALID_EXIT_CODE);
  assert (function != NULL);
  assert (strlen(function) < QMCKL_MAX_FUN_LEN);
  if (message != NULL) {
    assert (strlen(message)  < QMCKL_MAX_MSG_LEN);
  }

  if (qmckl_context_check(context) == QMCKL_NULL_CONTEXT)
    return QMCKL_INVALID_CONTEXT;

  if (message == NULL) {
    qmckl_exit_code rc =
      qmckl_set_error(context, exit_code, function, qmckl_string_of_error(exit_code));
    assert (rc == QMCKL_SUCCESS);
  } else {
    qmckl_exit_code rc =
      qmckl_set_error(context, exit_code, function, message);
    assert (rc == QMCKL_SUCCESS);
  }

  return exit_code;
}

* Last error

  ~qmckl_last_error~ formats the record into a caller-provided buffer of
  ~size_max~ bytes: ~Error -- <code> -- in <function>~ on one line, the
  message on the next. The text is truncated to fit and is always terminated.
  A ~NULL~ buffer and a non-positive size return ~QMCKL_INVALID_ARG_2~ and
  ~QMCKL_INVALID_ARG_3~. With an invalid context the function cannot read the
  record, writes ~"Null context"~ and returns ~QMCKL_FAILURE~.

  # Source
  #+begin_src c :tangle (eval c) :exports none
qmckl_exit_code
qmckl_last_error(qmckl_context context, char* buffer, const int64_t size_max) {

  char function_name[QMCKL_MAX_FUN_LEN];
  char message[QMCKL_MAX_MSG_LEN];

  qmckl_exit_code rc, last_rc;

  /* This function reports an error. It cannot record its own failure in the
     context without destroying the state it is about to report. */
  if (buffer == NULL) {
    return QMCKL_INVALID_ARG_2;
  }

  if (size_max <= 0) {
    return QMCKL_INVALID_ARG_3;
  }

  if (qmckl_context_check(context) == QMCKL_NULL_CONTEXT) {
    snprintf(buffer, (size_t) size_max, "Null context");
    return QMCKL_FAILURE;
  }

  rc = qmckl_get_error(context, &last_rc, function_name, message);
  if (rc != QMCKL_SUCCESS) {
    return rc;
  }

  /* The message is truncated when buffer is smaller than
     QMCKL_MAX_FUN_LEN + QMCKL_MAX_MSG_LEN + 64 bytes. */
  snprintf(buffer, (size_t) size_max, "Error -- %s -- in %s\n%s",
           qmckl_string_of_error(last_rc),
           function_name, message);

  return QMCKL_SUCCESS;
}

* Helper functions for debugging

  ~qmckl_check~ prints to ~stderr~ the code it was given and the record of the
  context when that code is not ~QMCKL_SUCCESS~. What it returns is the code
  it reads back from the context, not its argument: a matching context gives
  the same value, an empty one gives ~QMCKL_SUCCESS~, and the assertion that
  usually follows would then pass. The context must be valid; ~qmckl_get_error~
  asserts it.

  # Source
  #+begin_src c :tangle (eval c) :exports none
qmckl_exit_code
qmckl_check(qmckl_context context, qmckl_exit_code rc)
{

  if (rc != QMCKL_SUCCESS) {
    char fname[QMCKL_MAX_FUN_LEN];
    char message[QMCKL_MAX_MSG_LEN];

    fprintf(stderr, "===========\nQMCKL ERROR\n%s\n", qmckl_string_of_error(rc));
    qmckl_get_error(context, &rc, fname, message);
    fprintf(stderr, "Function: %s\nMessage: %s\n===========\n", fname, message);
  }

  return rc;
}

  /* Initialize the variables */
  char function_name[QMCKL_MAX_FUN_LEN]="";
  char message[QMCKL_MAX_MSG_LEN]="";

  /* Set the error code to be different from Success */
  qmckl_exit_code exit_code;
  exit_code = 1;

  assert (qmckl_set_error(context, exit_code, "my_function", "Message") == QMCKL_SUCCESS);

  assert (qmckl_get_error(context, &exit_code, function_name, message) == QMCKL_SUCCESS);
  assert (exit_code == 1);

  assert (strcmp(function_name,"my_function") == 0);
  assert (strcmp(message,"Message") == 0);

  /* qmckl_last_error must never write past the size it is given, and it
     must terminate its output. */
  {
    char last_error[QMCKL_MAX_FUN_LEN + QMCKL_MAX_MSG_LEN + 64];
    assert (qmckl_last_error(context, last_error, (int64_t) sizeof(last_error)) == QMCKL_SUCCESS);
    assert (strcmp(last_error,
                   "Error -- Invalid argument 1 -- in my_function\nMessage") == 0);

    char small[16];
    assert (qmckl_last_error(context, small, (int64_t) sizeof(small)) == QMCKL_SUCCESS);
    assert (strlen(small) == sizeof(small) - 1);

    assert (qmckl_last_error(context, NULL, (int64_t) sizeof(small)) == QMCKL_INVALID_ARG_2);
    assert (qmckl_last_error(context, small, 0) == QMCKL_INVALID_ARG_3);
  }

  exit_code = qmckl_context_destroy(context);
  assert(exit_code == QMCKL_SUCCESS);

  return 0;
}
