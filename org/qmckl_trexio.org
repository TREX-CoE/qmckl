* Introduction

  The [[https://trex-coe.github.io/trexio/trex.html][TREXIO library]] stores the data of a quantum-chemistry calculation —
  geometry, basis set, electron and orbital data — in a single file, in text
  or HDF5 format, so that codes can exchange it.

  The public function of this module, ~qmckl_trexio_read~, fills a context
  from such a file: it opens the file, reads the electrons, the nuclei, the
  atomic basis and the molecular orbitals, and closes it. The rest of this
  page documents the helpers that read one TREXIO group each.

  The file is tried as text first, then as HDF5. Effective core potentials are
  not read. When QMCkl is built without TREXIO, ~qmckl_trexio_read~ still
  exists but returns ~QMCKL_FAILURE~ with a message saying so.

* Local functions

  Functions defined in this section are all local (internal to this module):
  they should not be exposed in the public API. To identify them clearly,
  we append ~_X~ to their name, indicating their private status.

  Users are not able to call these functions directly. Since they are only
  called from within this module where the context validity has already been
  checked, the context parameter can't be ~NULL~. Therefore, we verify this
  precondition with an ~assert~ statement rather than returning an error code.

  In the functions defined in this section, we use the following local variable
  naming conventions:
  - ~rc~: the return code for QMCkl functions
  - ~rcio~: the return code for TREXIO library functions.

** Open file

   The file is opened once by a helper that tries the TEXT backend first and
   then the HDF5 backend, and returns a ~NULL~ pointer if both fail. The handle
   it returns is passed to the other helpers, each of which reads one group of
   data. ~qmckl_trexio_read~ closes it when it is done, on the error paths as
   well as on the last one.

   #+begin_src c :tangle (eval c) :exports none
#ifdef HAVE_TREXIO
trexio_t* qmckl_trexio_open_X(const char* file_name, qmckl_exit_code* rc)
{
  ,*rc = QMCKL_SUCCESS;
  trexio_t* file = NULL;

  file = trexio_open(file_name, 'r', TREXIO_TEXT, rc);
  if (file != NULL) return file;

  file = trexio_open(file_name, 'r', TREXIO_HDF5, rc);
  if (file != NULL) return file;

  ,*rc = QMCKL_FAILURE;

  return NULL;
}
#endif
   #+end_src

** Electron

   The electron group holds the number of up-spin and down-spin electrons.
   The helper reads both and sets them together with ~qmckl_set_electron_num~;
   no electron position is read.

   #+begin_src c :tangle (eval c) :exports none
#ifdef HAVE_TREXIO
qmckl_exit_code
qmckl_trexio_read_electron_X(qmckl_context context, trexio_t* const file)
{
  assert (context != (qmckl_context) 0);
  assert (file != NULL);

  int rcio = 0;

  int64_t up_num = 0L;
  int64_t dn_num = 0L;

  rcio = trexio_read_electron_up_num_64(file, &up_num);
  if (rcio != TREXIO_SUCCESS) {
    return qmckl_failwith( context,
                           QMCKL_FAILURE,
                           "trexio_read_electron_up_num",
                           trexio_string_of_error(rcio));
  }

  assert (up_num >= 0L);

  rcio = trexio_read_electron_dn_num_64(file, &dn_num);
  if (rcio != TREXIO_SUCCESS) {
    return qmckl_failwith( context,
                           QMCKL_FAILURE,
                           "trexio_read_electron_dn_num",
                           trexio_string_of_error(rcio));
  }

  assert (dn_num >= 0L);


  qmckl_exit_code rc;
  rc = qmckl_set_electron_num(context, up_num, dn_num);
  return rc;
}
#endif
   #+end_src

*** Number of nuclei

   ~nucleus.num~ is read and must be positive. It is set first, because the
   charges and the coordinates are stored per nucleus.

   #+begin_src c :tangle (eval c) :exports none

*** Nuclear charges

   The whole ~nucleus.charge~ array is read in one call and passed to

*** Nuclear coordinates

    The geometry comes from ~nucleus.coord~, which TREXIO stores as three
    coordinates per nucleus. The setter is called with ~'N'~, so QMCkl
    transposes the array into its own ~[3][nucl_num]~ layout.

   #+begin_src c :tangle (eval c) :exports none
  {
    qmckl_memory_info_struct mem_info = qmckl_memory_info_struct_zero;
    mem_info.size = nucleus_num * 3 * sizeof(double);

    double* nucl_coord = (double*) qmckl_malloc(context, mem_info);

    if (nucl_coord == NULL) {
      return qmckl_failwith( context,
                             QMCKL_ALLOCATION_FAILED,
                             "qmckl_trexio_read_nucleus_X",
                             NULL);
    }

    assert (nucl_coord != NULL);

    rcio = trexio_read_safe_nucleus_coord_64(file, nucl_coord, 3*nucleus_num);
    if (rcio != TREXIO_SUCCESS) {
      return qmckl_failwith( context,
                             QMCKL_FAILURE,
                             "trexio_read_nucleus_charge",
                             trexio_string_of_error(rcio));
    }

    rc = qmckl_set_nucleus_coord(context, 'N', nucl_coord, 3*nucleus_num);

    qmckl_free(context, nucl_coord);
    nucl_coord = NULL;

    if (rc != QMCKL_SUCCESS) {
      return rc;
    }
  }

** Basis set and AOs

   In this section we read the atomic basis set and atomic orbitals. The basis
   type and the three sizes come first, because every array that follows is
   allocated from one of the sizes. The per-nucleus and per-shell arrays come
   next, then the per-primitive arrays, and the AO normalization factor closes
   the group.

   #+begin_src c :tangle (eval c) :exports none
#ifdef HAVE_TREXIO
qmckl_exit_code
qmckl_trexio_read_ao_X(qmckl_context context, trexio_t* const file)
{
  assert (context != (qmckl_context) 0);
  assert (file != NULL);

  qmckl_exit_code rc;
  int rcio = 0;
  int64_t nucleus_num = 0L;

  rc = qmckl_get_nucleus_num(context, &nucleus_num);
  if (rc != QMCKL_SUCCESS)
    return rc;

   #+end_src

*** Basis set type

   ~basis.type~ is read as a string and only its first character is used. It
   must be ~G~ for a Gaussian basis or ~S~ for a Slater basis. Any other value
   fails: a leading ~N~ reaches ~qmckl_set_ao_basis_type~, which rejects it,
   and any other character is rejected here.

   #+begin_src c :tangle (eval c) :exports none
#define MAX_STR_LEN 1024
  char basis_type[MAX_STR_LEN];

  rcio = trexio_read_basis_type(file, basis_type, MAX_STR_LEN);
  if (rcio != TREXIO_SUCCESS) {
    return qmckl_failwith( context,
                           QMCKL_FAILURE,
                           "trexio_read_basis_type",
                           trexio_string_of_error(rcio));
  }

  if (basis_type[0] == 'G' ||
      basis_type[0] == 'S' ||
      basis_type[0] == 'N' ) {
    rc = qmckl_set_ao_basis_type(context, basis_type[0]);
  } else {
    return qmckl_failwith( context,
                           QMCKL_FAILURE,
                           "trexio_read_basis_type",
                           "Invalid basis type");
  }

  if (rc != QMCKL_SUCCESS)
    return rc;

   #+end_src

*** Nucleus index array

   ~basis.nucleus_index~ gives, for each shell, the nucleus it belongs to,
   numbered from zero. QMCkl keeps the opposite table, the index of the first
   shell of each nucleus. The helper walks the shells from the last to the
   first, so each nucleus ends up holding its first shell, and it rejects an
   index outside the range.

   #+begin_src c :tangle (eval c) :exports none
  {
    qmckl_memory_info_struct mem_info = qmckl_memory_info_struct_zero;

    /* Allocate array for data */
    mem_info.size = nucleus_num * sizeof(int64_t);
    int64_t* nucleus_index = (int64_t*) qmckl_malloc(context, mem_info);

    if (nucleus_index == NULL) {
      return qmckl_failwith( context,
                             QMCKL_ALLOCATION_FAILED,
                             "qmckl_trexio_read_basis_nucleus_index_X",
                             NULL);
    }

    assert (nucleus_index != NULL);

    /* Allocate temporary array */
    mem_info.size = shell_num * sizeof(int64_t);
    int64_t* tmp_array = (int64_t*) qmckl_malloc(context, mem_info);

    if (tmp_array == NULL) {
      qmckl_free(context, nucleus_index);
      nucleus_index = NULL;
      return qmckl_failwith( context,
                             QMCKL_ALLOCATION_FAILED,
                             "qmckl_trexio_read_basis_nucleus_index_X",
                             NULL);
    }

    assert (tmp_array != NULL);

    /* Read in the temporary array */
    rcio = trexio_read_safe_basis_nucleus_index_64(file, tmp_array, shell_num);
    if (rcio != TREXIO_SUCCESS) {
      qmckl_free(context, tmp_array);
      tmp_array = NULL;
      qmckl_free(context, nucleus_index);
      nucleus_index = NULL;
      return qmckl_failwith( context,
                             QMCKL_FAILURE,
                             "trexio_read_basis_nucleus_index",
                             trexio_string_of_error(rcio));
    }

    /* Reformat data */
    for (int i=shell_num-1 ; i>=0 ; --i) {
      const int k = tmp_array[i];
      if (k < 0 || k >= nucleus_num) {
        qmckl_free(context, tmp_array);
        tmp_array = NULL;
        qmckl_free(context, nucleus_index);
        nucleus_index = NULL;
        return qmckl_failwith( context,
                                QMCKL_FAILURE,
                                "trexio_read_basis_nucleus_index",
                                "Irrelevant data in TREXIO file");
      }
      nucleus_index[k] = i;
    }

    qmckl_free(context, tmp_array);
    tmp_array = NULL;

    /* Store data */
    rc = qmckl_set_ao_basis_nucleus_index(context, nucleus_index, shell_num);

    qmckl_free(context, nucleus_index);
    nucleus_index = NULL;

    if (rc != QMCKL_SUCCESS)
      return rc;
  }
   #+end_src

   The same dataset is read a second time and counted per nucleus, which gives
   ~nucleus_shell_num~.

      nucleus_shell_num = NULL;

    }

    qmckl_free(context, tmp_array);
    tmp_array = NULL;

    /* Store data */

    qmckl_free(context, nucleus_shell_num);
    nucleus_shell_num = NULL;

    if (rc != QMCKL_SUCCESS)

   ~basis.shell_index~ is read a second time and walked backward over the
   primitives, as ~nucleus_index~ was walked backward over the shells. What

* Read everything

  Reads the electron, nucleus, atomic-basis and molecular-orbital data of a
  TREXIO file into the context, in that order, and closes the file. A NULL
  context is rejected with ~QMCKL_INVALID_CONTEXT~; a NULL ~file_name~ and an
  out-of-range ~size_max~ are rejected before the file is opened, with
  ~QMCKL_INVALID_ARG_2~ and ~QMCKL_INVALID_ARG_3~.

  ~size_max~ is the number of characters that may be read from ~file_name~,
  excluding the terminating NUL character. It must be positive and smaller
  than 4096, the size of the buffer that holds the NUL-terminated copy of the
  name. When the file cannot be opened with either backend, the read fails
  with ~QMCKL_INVALID_ARG_2~.

  #+begin_src c :tangle (eval h_func)
qmckl_exit_code
qmckl_trexio_read(const qmckl_context context,
                  const char* file_name,
                  const int64_t size_max);
  #+end_src

  #+begin_src f90 :tangle (eval fh_func) :comments org :exports none

  #+begin_src c :tangle (eval c) :exports none
/* PATH_MAX is 4096 on Linux and 1024 on macOS. A longer argument cannot
   name a file this library is able to open. */
#define QMCKL_TREXIO_MAX_FILE_NAME 4096

qmckl_exit_code
qmckl_trexio_read(const qmckl_context context, const char* file_name, const int64_t size_max)
{
  if (qmckl_context_check(context) == QMCKL_NULL_CONTEXT) {
    return QMCKL_INVALID_CONTEXT;
  }

  if (file_name == NULL) {
    return qmckl_failwith( context,
                           QMCKL_INVALID_ARG_2,
                           "qmckl_trexio_read",
                           "file_name is a null pointer");
  }

  if (size_max <= 0 || size_max >= QMCKL_TREXIO_MAX_FILE_NAME) {
    return qmckl_failwith( context,
                           QMCKL_INVALID_ARG_3,
                           "qmckl_trexio_read",
                           "size_max out of range");
  }

  qmckl_exit_code rc;
  char file_name_new[QMCKL_TREXIO_MAX_FILE_NAME];
  strncpy(file_name_new, file_name, (size_t) size_max);
  file_name_new[(size_t) size_max] = '\0';

#ifdef HAVE_TREXIO
  trexio_t* file = qmckl_trexio_open_X(file_name_new, &rc);
  if (file == NULL) {
    return qmckl_failwith( context,
                           QMCKL_INVALID_ARG_2,
                           "qmckl_trexio_read",
                           trexio_string_of_error(rc));
  }

  assert (file != NULL);

  rc = qmckl_trexio_read_electron_X(context, file);
  if (rc != QMCKL_SUCCESS) {
    trexio_close(file);
    return rc;
  }

  rc = qmckl_trexio_read_nucleus_X(context, file);
  if (rc != QMCKL_SUCCESS) {
    trexio_close(file);
    return rc;
  }

  rc = qmckl_trexio_read_ao_X(context, file);
  if (rc != QMCKL_SUCCESS) {
    trexio_close(file);
    return rc;
  }

  rc = qmckl_trexio_read_mo_X(context, file);
  if (rc != QMCKL_SUCCESS) {
    trexio_close(file);
    return rc;
  }

  trexio_close(file);
  file = NULL;
#else

  rc = qmckl_failwith( context,
                       QMCKL_FAILURE,
                       "qmckl_trexio_read",
                       "QMCkl was compiled without TREXIO");
#endif
  return rc;
}
